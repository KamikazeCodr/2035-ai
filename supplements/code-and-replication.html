def run_simulation(N, SEED=7, cap_fraud_mult=5.0):

    """
    Execute calibrated Monte Carlo simulation for N samples.
    Enforces dynamic defense reference points and macroeconomic ceilings.
    """

    for s in P.CAL:

        P.CAL[s] = 1.0



    pilot = V.run(N, SEED, {**C.SW, "S4def": False, "S4dep": False})

    w0 = C.weights(pilot)

    ref = {(s, y): wq(pilot["L"][s][y], w0, (0.5,))[0] for s in SECT for y in YEARS}



    out = V.run(N, SEED, C.SW, ref=ref, keep=(2025, 2026, 2030, 2035))

    w = C.weights(out)

    base, bess = C.baseline_post(N, SEED)



    ceiling = {y: 3450.0 * 1.03 ** (y - 2025) for y in YEARS}

    tb = {y: V.total(out, "b", y) for y in YEARS}



    # Cap fraud growth tail post-2026 baseline calibration

    if cap_fraud_mult is not None:

        b_fraud_2025 = out["b"]["fraud"][2025]

        b_fraud_2026 = out["b"]["fraud"][2026]

        cap_val = cap_fraud_mult * b_fraud_2025

        clipped_fraud_2026 = np.minimum(b_fraud_2026, cap_val)

        tb[2026] = tb[2026] - b_fraud_2026 + clipped_fraud_2026



    ta = {y: base[y] + tb[y] for y in YEARS}



    # Boundary assertions and empirical support bounds enforcement

    raw_violations = {y: int(np.sum(tb[y] > ta[y])) for y in YEARS}

    ta_trunc = {y: np.minimum(ta[y], ceiling[y]) for y in YEARS}

    p_above_ceil = {y: float((w * (ta[y] > ceiling[y])).sum()) for y in YEARS}



    # Macroeconomic ceiling enforcement

    clipped_violations = {y: int(np.sum(tb[y] > ta_trunc[y])) for y in YEARS}

    tb_clipped = {y: np.minimum(tb[y], ta_trunc[y]) for y in YEARS}



    # Quantiles: p10, p25, p50, p75 across calibrated distributions

    q_b = {

        y: dict(zip(("p10", "p25", "p50", "p75"), wq(tb_clipped[y], w, (0.1, 0.25, 0.5, 0.75))))

        for y in YEARS

    }

    q_a = {

        y: dict(zip(("p10", "p25", "p50", "p75"), wq(ta_trunc[y], w, (0.1, 0.25, 0.5, 0.75))))

        for y in YEARS

    }

    frac_b_gt_half_a = {y: float((w * (tb_clipped[y] / ta_trunc[y] > 0.5)).sum()) for y in YEARS}

    ]

    post_w = {names[i]: float(w[br == i].sum()) for i in range(5)}



    branch_b = {}

    branch_a = {}

            wm = w[m] / w[m].sum()

            branch_b[nm] = {

                y: dict(zip(("p10", "p25", "p50", "p75"), wq(tb_clipped[y][m], wm, (0.1, 0.25, 0.5, 0.75))))

            }

            branch_a[nm] = {

                y: dict(zip(("p10", "p25", "p50", "p75"), wq(ta_trunc[y][m], wm, (0.1, 0.25, 0.5, 0.75))))

                for y in (2026, 2030, 2035)

            }



    return {

        "N": N,

        "ess": ess(w),

loss_mc.py (Modular Loss Engine) 
#!/usr/bin/env python3

r"""
Probabilistic Monte Carlo Cyber Loss Engine (2024-2035)
Author: Aditya Singh (refined with Sonnet)

Formulates global cyber loss exposure as a multi-sector product of empirical operational factors:
    L_total(t) = L_base(t) + \Delta L_AI(t) = L_base(t) + \left( \sum_{s \in \mathcal{S}} \prod_{k=1}^7 f_{k,s}(t) \right) \times P_{33}

Samples parameter uncertainty over lognormal and logit margins using an empirical
macro-shock correlation structure (rho = 0.30) to capture systemic crisis clustering across sectors.
Includes Spearman rank correlation sensitivity evaluations and automated boundary assertions.
"""

import sys

import json

import numpy as np



Z90 = 1.2815515655446004  # Standard normal 90th percentile





def template():

    """Generate empty template specification dictionary for factor-product modeling."""

    years = [2027, 2028]

    q = {"p10": None, "p50": None, "p90": None, "anchor": None}

    f = {"kind": "lognormal", "by_year": {str(y): dict(q) for y in years}}

    return {

        "metric": "Empirical cyber loss exposure",

        "unit": "USD bn per year, constant 2025 USD",

        "years": years,

        "n_samples": 20000,

        "seed": 0,

        "rho": 0.3,

        "factors": {

            "capable_actors": dict(f),

            "campaigns_per_actor": dict(f),

            "exposed_share": {**f, "kind": "fraction"},

            "success_given_exposure": {**f, "kind": "fraction"},

            "loss_per_success": dict(f),

            "defense_containment": {**f, "kind": "complement"},

            "reporting_multiplier": dict(f),

        },

        "reference": {"2034": None},

    }





class ValidationError(ValueError):

    """Raised when simulation parameters or sector factor specifications fail validation checks."""

    pass





def validate(spec):

    """
    Validate input factor specification dictionary for mathematical consistency.
    Raises ValidationError if any parameter or quantile is malformed.
    """

    if "factors" not in spec:

        raise ValidationError("Missing 'factors' dictionary in specification")



    if "years" not in spec or not spec["years"]:

        raise ValidationError("Missing or empty 'years' list in specification")



    for name, fac in spec["factors"].items():

        if fac.get("kind") not in ("lognormal", "fraction", "complement"):

            raise ValidationError(f"{name}: unknown factor distribution kind '{fac.get('kind')}'")



        by_year = fac.get("by_year", {})

        for y in spec["years"]:

            q = by_year.get(str(y))

            if not q or any(q.get(k) is None for k in ("p10", "p50", "p90")):

                raise ValidationError(f"{name} {y}: p10/p50/p90 quantiles missing")



            if not q.get("anchor"):

                raise ValidationError(

                    f"{name} {y}: missing 'anchor' citation (source citation or methodological rationale required)"

                )



            if not (q["p10"] <= q["p50"] <= q["p90"]):

                raise ValidationError(

                    f"{name} {y}: non-monotonic quantiles (expected p10 <= p50 <= p90, got {q['p10']} / {q['p50']} / {q['p90']})"

                )



            if fac["kind"] != "lognormal" and not (0 < q["p10"] and q["p90"] < 1):

                raise ValidationError(f"{name} {y}: fraction values must lie strictly in (0, 1)")



            if fac["kind"] == "lognormal" and q["p10"] <= 0:

                raise ValidationError(f"{name} {y}: lognormal requires p10 > 0")





def draw(q, kind, z):

    """Draw samples from marginal quantile distribution given standard normal shocks z."""

    if kind == "lognormal":

        mu = np.log(q["p50"])

        sig = (np.log(q["p90"]) - np.log(q["p10"])) / (2.0 * Z90)

        return np.exp(mu + sig * z)



    mu = np.log(q["p50"] / (1.0 - q["p50"]))

    sig = (np.log(q["p90"] / (1.0 - q["p90"])) - np.log(q["p10"] / (1.0 - q["p10"]))) / (2.0 * Z90)



    if kind == "complement":

        # A positive shock z represents an unfavorable world for defense (reduced containment),

        # so containment fraction is drawn with -z, and defensive factor is (1 - containment).

        p = 1.0 / (1.0 + np.exp(-(mu - sig * z)))

        return 1.0 - p



    return 1.0 / (1.0 + np.exp(-(mu + sig * z)))





def spearman(a, b):

    """Compute Spearman rank correlation between two sample vectors."""

    ra = np.argsort(np.argsort(a))

    rb = np.argsort(np.argsort(b))

    return float(np.corrcoef(ra, rb)[0, 1])





def main():

    """Command-line interface for loss_mc module."""

    if len(sys.argv) < 2:

        print(__doc__)

        return 1



    if sys.argv[1] == "--template":

        print(json.dumps(template(), indent=2))

        return 0



    with open(sys.argv[1], "r", encoding="utf-8") as f:

        spec = json.load(f)



    validate(spec)

    n = spec["n_samples"]

    rho = spec["rho"]

    rng = np.random.default_rng(spec["seed"])

    names = list(spec["factors"])

    years = spec["years"]



    results = {}

    final_parts = {}



    for y in years:

        common = rng.standard_normal(n)

        total = np.ones(n)

        for name in names:

            fac = spec["factors"][name]

            z = np.sqrt(rho) * common + np.sqrt(1.0 - rho) * rng.standard_normal(n)

            part = draw(fac["by_year"][str(y)], fac["kind"], z)

            total *= part

            if y == years[-1]:

                final_parts[name] = part

        results[y] = total



    out = {

        "metric": spec["metric"],

        "unit": spec["unit"],

        "rho": rho,

        "years": {},

    }



    for y in years:

        p5, p10, p50, p90, p95 = np.percentile(results[y], [5, 10, 50, 90, 95])

        out["years"][str(y)] = {

            "p5": float(p5),

            "p10": float(p10),

            "p50": float(p50),

            "p90": float(p90),

            "p95": float(p95),

        }



    print(json.dumps(out, indent=2))

    return 0





if __name__ == "__main__":

    sys.exit(main())

def logit(p):

    """Logit transform."""

    return math.log(p / (1.0 - p))





def inv_logit(x):

    """Inverse logit transform."""

    return 1.0 / (1.0 + math.exp(-x))

    check(

    hand = 40e3 * 6e3 * P.CAL["fraud"] * 0.010 * 0.010 * 4.8e-5 * (1.0 - 0.12) * 7.0

    def prod_med(spec, y):

        t = 1.0

        for fn, f in spec["factors"].items():

            p = f["by_year"][str(y)]["p50"]

            t *= (1.0 - p) if f["kind"] == "complement" else p

        return t

    # Independent re-implementation verification across representative sector-years

    Z_val = 1.2815515655446004



    def independent_sample(sector, year, sample_size, sample_seed):

        local_rng = np.random.default_rng(sample_seed)

        w_factor = 1.0 + 0.03 * max(0, year - 2026)

        macro_shock = local_rng.standard_normal(sample_size)

        accumulated = np.ones(sample_size)



        for factor_name, f_spec in P.SECTORS[sector].items():

            ks = P.KEY

            i_idx = 0 if year <= ks[0] else (len(ks) - 2 if year >= ks[-1] else max(j for j in range(len(ks)) if ks[j] <= year))

            t_weight = (year - ks[i_idx]) / (ks[i_idx + 1] - ks[i_idx])

            v_p50 = f_spec["p50"]

            z_factor = math.sqrt(0.3) * macro_shock + math.sqrt(0.7) * local_rng.standard_normal(sample_size)



            if f_spec["kind"] == "lognormal":

                mu_val = math.log(v_p50[i_idx]) + t_weight * (math.log(v_p50[i_idx + 1]) - math.log(v_p50[i_idx]))

                if factor_name == "campaigns_per_actor":

                    mu_val += math.log(P.CAL[sector])

                sig_val = w_factor * (math.log(f_spec["spread"][0]) + math.log(f_spec["spread"][1])) / (2.0 * Z_val)

                accumulated *= np.exp(mu_val + sig_val * z_factor)

            else:

                mu_val = logit(v_p50[i_idx]) + t_weight * (logit(v_p50[i_idx + 1]) - logit(v_p50[i_idx]))

                sig_val = w_factor * (f_spec["spread"][0] + f_spec["spread"][1]) / (2.0 * Z_val)

                sign_val = -1 if f_spec["kind"] == "complement" else 1

                prob = 1.0 / (1.0 + np.exp(-(mu_val + sign_val * sig_val * z_factor)))

                accumulated *= (1.0 - prob) if f_spec["kind"] == "complement" else prob



        return accumulated



    n_indep = 1_000_000

    R.N = n_indep

    for s_name, y_val in (("fraud", 2026), ("it", 2030), ("ics", 2035)):

        sim_a = independent_sample(s_name, y_val, n_indep, 5)

        specs_model = R.spec_for("B-03", s_name)

        rr = np.random.default_rng(6)

        z0 = rr.standard_normal(n_indep)

        sim_b = np.ones(n_indep)

        for fn_name, fac_spec in specs_model["factors"].items():

            zf = math.sqrt(0.3) * z0 + math.sqrt(0.7) * rr.standard_normal(n_indep)

            sim_b *= M.draw(fac_spec["by_year"][str(y_val)], fac_spec["kind"], zf)



        quantiles_a = [np.percentile(sim_a, p) for p in (10, 50, 90)]

        quantiles_b = [np.percentile(sim_b, p) for p in (10, 50, 90)]

        dev_list = [qa / qb - 1.0 for qa, qb in zip(quantiles_a, quantiles_b)]

        check(

            f"Independent re-implementation {s_name} {y_val}: p10/p50/p90 match within 3%",

            max(abs(x) for x in dev_list) < 0.03,

            "deviations: " + ", ".join(f"{x:+.2%}" for x in dev_list),

        )
